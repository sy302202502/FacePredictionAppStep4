SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# ── 週次予想パイプライン（毎朝9時: 出馬表→統計予想→顔面分析→重賞検証）──
0 9 * * * cd /opt/faceprediction && docker compose exec -T python python3 python/weekly_pipeline.py >> logs/weekly_cron.log 2>&1

# ── レース当日の出馬表同期（土日 11/13/15時: 取消馬の除外・馬番確定）──
0 11,13,15 * * 6,0 cd /opt/faceprediction && docker compose exec -T python python3 python/entry_fetcher.py --sync >> logs/sync_cron.log 2>&1

# ── レース当日のオッズ更新（土日 9/11/13/15時: predict-v2のオッズバッジ用）──
0 9,11,13,15 * * 6,0 cd /opt/faceprediction && docker compose exec -T python python3 python/odds_fetcher.py >> logs/odds_cron.log 2>&1

# ── 的中記録の自動取得（土日夜 + 毎朝10時: 未記録分のみ処理・冪等）──
0 18,22 * * 6,0 cd /opt/faceprediction && docker compose exec -T python python3 python/result_auto_fetcher.py >> logs/results_cron.log 2>&1
0 10 * * * cd /opt/faceprediction && docker compose exec -T python python3 python/result_auto_fetcher.py >> logs/results_cron.log 2>&1

# ── 本番DBの日次バックアップ（4時: 14日保持・空ダンプ検知付き）──
0 4 * * * /opt/faceprediction/deploy/vps_backup.sh >> /opt/faceprediction/logs/backup.log 2>&1

# ── uploads の古い画像GC（月曜5時: race_specific=180日/paddock=30日）──
0 5 * * 1 cd /opt/faceprediction && docker compose exec -T python python3 python/cleanup_uploads.py --apply >> logs/cleanup.log 2>&1

# ── 古いcronログ削除（毎日3時: 30日以前）──
0 3 * * * find /opt/faceprediction/logs -name "*_cron.log*" -mtime +30 -delete
